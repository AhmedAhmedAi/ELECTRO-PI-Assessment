> Write a Python function merge_sorted(a, b) that merges two already-sorted lists into one sorted list in O(len(a)+len(b)) time, without using sorted() or sort(). Include two doctest examples.
```python
def merge_sorted(a, b):
    """
    Merge two already-sorted lists into one sorted list in O(len(a)+len(b)) time.
    
    Args:
        a (list): First sorted list.
        b (list): Second sorted list.
    
    Returns:
        list: A new sorted list containing all elements from a and b.
    """
    result = []
    i, j = 0, 0
    len_a, len_b = len(a), len(b)

    # Merge by comparing elements from both lists
    while i < len_a and j < len_b:
        if a[i] <= b[j]:
            result.append(a[i])
            i += 1
        else:
            result.append(b[j])
            j += 1

    # Append any remaining elements from a
    while i < len_a:
        result.append(a[i])
        i += 1

    # Append any remaining elements from b
    while j < len_b:
        result.append(b[j])
        j += 1

    return result


# Doctest examples
if __name__ == "__main__":
    # Example 1: Basic merge
    >>> merge_sorted([1, 3, 5], [2, 4, 6])
    [1, 2, 3, 4, 5, 6]

    # Example 2: One list is empty
    >>> merge_sorted([], [1, 2, 3])
    [1, 2, 3]
    >>> merge_sorted([1, 2, 3], [])
    [1, 2, 3]
```
